    const chartOfAccounts = {
        "606400": "Fournitures administratives",
        "401100": "Fournisseurs divers",
        "445660": "TVA déductible sur ABS",
        "707000": "Ventes de marchandises",
        "411100": "Clients divers",
        "445710": "TVA collectée",
        "512000": "Banque"
    };

    let rowCount = 0;

    function addRow(account = "", label = "", debit = "", credit = "") {
        rowCount++;
        const tbody = document.getElementById("tableBody");
        const tr = document.createElement("tr");
        tr.id = `row-${rowCount}`;

        tr.innerHTML = `
            <td>
                <input type="text" class="acc-input" list="accountsList" value="${account}" onchange="updateAccountLabel(this)" placeholder="N° Compte">
                <datalist id="accountsList">
                    ${Object.keys(chartOfAccounts).map(code => `<option value="${code}">${code} -${chartOfAccounts[code]}</option>`).join('')}
                </datalist>
            </td>
            <td><input type="text" class="label-input" value="${label}"></td>
            <td><input type="number" step="0.01" class="debit-input" value="${debit}" oninput="handleAmountInput(this, 'debit')"></td>
            <td><input type="number" step="0.01" class="credit-input" value="${credit}" oninput="handleAmountInput(this, 'credit')"></td>
            <td style="text-align:center;"><button class="btn btn-danger" onclick="removeRow('row-${rowCount}')">X</button></td>
        `;

        tbody.appendChild(tr);
        recalculateTotals();
    }

    function removeRow(rowId) {
        document.getElementById(rowId).remove();
        recalculateTotals();
    }

    function updateAccountLabel(element) {
        const row = element.closest("tr");
        const accCode = element.value;
        const labelInput = row.querySelector(".label-input");
        
        if (chartOfAccounts[accCode] && !labelInput.value) {
            const generalLabel = document.getElementById("libelleGen").value;
            labelInput.value = generalLabel || chartOfAccounts[accCode];
        }
    }

    function handleAmountInput(element, type) {
        const row = element.closest("tr");
        const oppositeInput = row.querySelector(type === 'debit' ? '.credit-input' : '.debit-input');
        if (element.value !== "") {
            oppositeInput.value = "";
        }
        recalculateTotals();
    }
